06 / Core React idea

The sprite sheet behaves like an icon component 
const POSES = {
  angry: 0, cheer: 1, love: 2, running: 3,
  thumbsUp: 4, surprised: 5, sleepy: 6, dancing: 7,
  wave: 8, confused: 9, sad: 10, flex: 11
};

function BulldogSprite({ pose = "wave", size = 96 }) {
  const index = POSES[pose];
  const col = index % 4;
  const row = Math.floor(index / 4);

  return <span style={{
    width: size,
    height: size,
    backgroundImage: 'url("/assets/bulldog-sprite-sheet.png")',
    backgroundSize: "400% 300%",
    backgroundPosition: `${(col / 3) * 100}% ${(row / 2) * 100}%`,
    imageRendering: "pixelated"
  }} />;
}

// Then anywhere in the app:
<BulldogSprite pose="thumbsUp" size={96} />
